## s04_bands.py

```python
# 第一层预激活 z1 按时间偏移分带分解：各带贡献的方差、与中心帧标签的关联(η²)，并与"同样能量的随机权重"对照
import numpy as np
from tools import *
w = load_net('subject_net.npz'); dA = load_A()
rng = np.random.default_rng(0)
idx = np.sort(rng.choice(len(dA), 60000, replace=False))
X = dA.windows(idx).reshape(-1, 81, 28); y = dA.y[idx]
W1 = w['W1'].reshape(81, 28, 1024)
z1 = X.reshape(len(idx), -1) @ w['W1'] + w['b1']
def eta2(c, y):   # 每列：类间方差 / 总方差，返回各单元的平均
    tot = c.var(0); m = c.mean(0); bw = np.zeros(c.shape[1])
    for k in np.unique(y):
        s = y == k; bw += s.mean() * (c[s].mean(0) - m) ** 2
    return bw / tot
bands = [(0, 0), (1, 3), (4, 8), (9, 15), (16, 20), (21, 25), (26, 30), (31, 35), (36, 40)]
vz = z1.var(0)
print('z1: mean of per-unit var %.3f ; eta2(z1) %.3f ; active frac %.3f' % (vz.mean(), eta2(z1, y).mean(), (z1 > 0).mean()))
for lo, hi in bands:
    ts = [40 + d for d in range(-hi, hi + 1) if lo <= abs(d) <= hi]
    c = np.einsum('ntf,tfj->nj', X[:, ts], W1[ts])
    # 对照：把这些带的权重换成同 rms 的高斯随机权重
    R = rng.standard_normal(W1[ts].shape).astype(np.float32) * np.sqrt((W1[ts] ** 2).mean())
    cr = np.einsum('ntf,tfj->nj', X[:, ts], R)
    cov_rest = ((c - c.mean(0)) * (z1 - c - (z1 - c).mean(0))).mean(0)
    print(f'|dt| {lo:2d}-{hi:2d}: var share {np.mean(c.var(0)/vz):.3f}  eta2 {eta2(c, y).mean():.4f}  (random-weight control eta2 {eta2(cr, y).mean():.4f}, var share {np.mean(cr.var(0)/vz):.3f})  corr with rest {np.mean(cov_rest/np.sqrt(c.var(0)*(z1-c).var(0))):.3f}')
```

## s06_offsets_strat.py

```python
# (a) 每个时间偏移单独的 η²（学到的权重 vs 同能量随机权重），找信号消失的位置
# (b) 只看原网络：远处噪声贡献越大的帧，原网络是不是错得越多（分层分析，避开句首尾补零区）
import numpy as np
from tools import *
w = load_net('subject_net.npz'); dA = load_A()
W1 = w['W1'].reshape(81, 28, 1024)
rng = np.random.default_rng(2)
fs = np.concatenate([[0], np.cumsum(dA.lens)[:-1]]); pos = np.arange(len(dA)) - fs[dA.utt]; rpos = dA.lens[dA.utt] - 1 - pos
mid = np.where((pos >= 40) & (rpos >= 40))[0]
idx = np.sort(rng.choice(mid, 80000, replace=False))
X = dA.windows(idx).reshape(-1, 81, 28); y = dA.y[idx]
def eta2(c, y):
    tot = c.var(0); m = c.mean(0); bw = np.zeros(c.shape[1])
    for k in np.unique(y):
        s = y == k; bw += s.mean() * (c[s].mean(0) - m) ** 2
    return bw / tot
print('|dt|  eta2_learned  eta2_random  ratio   (two sides pooled, dims 0-27)')
for d in [12, 15, 18, 20, 22, 24, 26, 28, 30, 33, 36, 38, 40]:
    ts = [40 - d, 40 + d]
    c = np.einsum('ntf,tfj->nj', X[:, ts], W1[ts])
    e_r = []
    for rep in range(3):
        R = rng.standard_normal(W1[ts].shape).astype(np.float32) * np.sqrt((W1[ts] ** 2).mean())
        e_r.append(eta2(np.einsum('ntf,tfj->nj', X[:, ts], R), y).mean())
    print(f'{d:3d}   {eta2(c, y).mean():.4f}       {np.mean(e_r):.4f}     {eta2(c, y).mean()/np.mean(e_r):.2f}')
# (b) 分层
lg0 = np.load('cache_logits0.npy')[idx]; ok = lg0.argmax(1) == y
far = [t for t in range(81) if abs(t - 40) >= 26]
cfar = np.einsum('ntf,tfj->nj', X[:, far], W1[far])
z1 = X.reshape(len(idx), -1) @ w['W1'] + w['b1']
# 噪声相对强度：远处贡献的能量 / 总预激活能量（按帧）
r = (cfar ** 2).sum(1) / ((z1 - w['b1']) ** 2).sum(1)
R = rng.standard_normal(W1[far].shape).astype(np.float32) * np.sqrt((W1[far] ** 2).mean())
rr = (np.einsum('ntf,tfj->nj', X[:, far], R) ** 2).sum(1) / ((z1 - w['b1']) ** 2).sum(1)
q = np.quantile(r, [0, .2, .4, .6, .8, 1]); qr = np.quantile(rr, [0, .2, .4, .6, .8, 1])
print('far-noise energy share quintiles (learned weights):', np.round(q, 3))
print(' acc by quintile:', [round(ok[(r >= q[i]) & (r <= q[i+1])].mean(), 4) for i in range(5)])
print(' control (random far weights) acc by quintile:', [round(ok[(rr >= qr[i]) & (rr <= qr[i+1])].mean(), 4) for i in range(5)])
# 在同一 rr 分层内，再按 r 高低分（控制"远处输入能量/近处能量"这一混杂）
print(' within control-quintile: acc (low r half, high r half):')
for i in range(5):
    s = (rr >= qr[i]) & (rr <= qr[i+1]); med = np.median(r[s])
    print('   ', round(ok[s & (r <= med)].mean(), 4), round(ok[s & (r > med)].mean(), 4))
```

print('zeroed time offsets:', [t - 40 for t in far])
W1[far] = 0; w['W1'] = W1.reshape(2268, 1024)
# 原网络：h1、h2 的全量均值、h2 是否激活过、平均 softmax（校准检查，原网络允许）
s_h1_0 = np.zeros(1024); s_h2_0 = np.zeros(1024); ever = np.zeros(1024, bool); s_h1_1 = np.zeros(1024); s_sm = np.zeros(42)
for idx in batches():
    a = forward(w0, dA, idx)
    s_h1_0 += a['h1'].sum(0); s_h2_0 += a['h2'].sum(0); ever |= (a['h2'] > 0).any(0)
    lg = a['logits']; e = np.exp(lg - lg.max(1, keepdims=True)); s_sm += (e / e.sum(1, keepdims=True)).sum(0)
    x = a['x']; s_h1_1 += np.maximum(x @ w['W1'] + w['b1'], 0).sum(0)
Eh1_0, Eh2_0, Eh1_1 = s_h1_0 / N, s_h2_0 / N, s_h1_1 / N
freq = np.bincount(y, minlength=42) / N
print('calibration (orig): mean softmax vs label freq, max abs log-ratio over classes 0-39: %.3f' % np.abs(np.log(s_sm[:40] / N / freq[:40])).max())
print('  per class log(mean softmax / freq):', np.round(np.log(s_sm[:40] / N / freq[:40]), 2))
print('h1 mean: orig %.4f  after edit1 %.4f' % (Eh1_0.mean(), Eh1_1.mean()))
# 修改2：b2 补偿第一层均值变化，使 z2 的均值不变
w['b2'] = w['b2'] + (Eh1_0 - Eh1_1) @ w['W2']
print('b2 shift: mean %.4f  max abs %.4f' % (((Eh1_0 - Eh1_1) @ w['W2']).mean(), np.abs((Eh1_0 - Eh1_1) @ w['W2']).max()))
# 修改3：在整个 A 上从未激活过的 h2 单元，其 W3 行（仍是初始化尺度的噪声）置 0
dead = ~ever; print('h2 units never active on all of A:', dead.sum())
w['W3'][dead] = 0
# 修改4：b3 补偿第二层均值变化
s_h2_1 = np.zeros(1024); ever1 = np.zeros(1024, bool)
for idx in batches():
    h2 = forward(w, dA, idx)['h2']; s_h2_1 += h2.sum(0); ever1 |= (h2 > 0).any(0)
Eh2_1 = s_h2_1 / N
w['b3'] = w['b3'] + (Eh2_0 - Eh2_1) @ w['W3']
print('h2 mean: orig %.4f after %.4f ; units active after edit (on A): %d (orig %d)' % (Eh2_0.mean(), Eh2_1.mean(), ever1.sum(), ever.sum()))
print('b3 shift:', np.round((Eh2_0 - Eh2_1) @ w['W3'], 3))
# 修改5：[SOS]/[EOS] 从不作为标签，压低其偏置
w['b3'][40] = w['b3'][41] = -50.0
save_net(w, 'edited_net.npz'); print('saved edited_net.npz')
