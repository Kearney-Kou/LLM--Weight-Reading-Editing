# 诊断 5：每个特征在不同时间滞后上的类别区分度（看是否有维度被错位）；f23 是否掺了噪声；f13/f14 的关系

from tools import *

x = dA.P[dA.center]; cnt = np.bincount(y, minlength=40)[:40]
def eta2_at(f, lag):
    # 第 t 帧的标签 vs 第 t+lag 帧的特征 f（同一句内）
    s = dA.utt[max(lag, 0): N + min(lag, 0)] == dA.utt[max(-lag, 0): N - max(lag, 0)]
    yy = y[max(-lag, 0): N - max(lag, 0)][s]; xx = x[max(lag, 0): N + min(lag, 0), f][s]
    mu = np.bincount(yy, xx, 40) / np.maximum(np.bincount(yy, minlength=40), 1)
    return (np.bincount(yy, minlength=40) * mu ** 2).sum() / len(yy) / xx.var()
lags = list(range(-6, 7))

## edits.py（修改函数库，被后续脚本 import）

```python
# 修改函数库（每个函数只做一处有明确理由的修改）
import numpy as np
from tools import *

def copy(w): return {k: v.copy() for k, v in w.items()}

def drop_noise_feature(w, dA, f=26, compensate=True, n=40000, seed=0):
    """f=26 是纯白噪声（自相关≈0、类别区分度≈0、峰度≈3），第一层在它上面的权重只会往 z1 注入噪声。
    把 W1 中所有 t 的 f=26 行置 0。compensate=True 时，再把“噪声经 ReLU 凸性抬高的 h1 均值”
    折算进 b2，使第二层看到的输入均值与原网络一致（只用无标签的激活均值）。"""
    w2 = copy(w); rows = np.arange(81) * 28 + f
    w2['W1'][rows] = 0
    if compensate:
        rng = np.random.default_rng(seed); idx = np.sort(rng.choice(len(dA), n, replace=False))
        d = forward(w, dA, idx)['h1'].mean(0) - forward(w2, dA, idx)['h1'].mean(0)
        w2['b2'] = w2['b2'] + d @ w['W2']
    return w2

def prior_correct(w, logits, y):
    """一次性把每类输出偏置平移 log(类别频率 / 平均预测概率)，修正输出层偏置的先验错配；
    [SOS]/[EOS] 从不作为标签出现，偏置设为 -20。只做一步，不迭代。"""
    w2 = copy(w)
    pr = np.exp(logits - logits.max(1, keepdims=True)); pr /= pr.sum(1, keepdims=True)
    qbar = pr.mean(0); pi = np.bincount(y, minlength=42) / len(y)
    w2['b3'][:40] += np.log(pi[:40] / qbar[:40]).astype(np.float32)
    w2['b3'][40:] = -20.0
    return w2
```
