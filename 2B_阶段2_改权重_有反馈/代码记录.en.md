## edits.py (library of edit functions, imported by later scripts)

```python
# Library of edit functions (each function makes only one edit, with a clear reason)
import numpy as np
from tools import *

def copy(w): return {k: v.copy() for k, v in w.items()}

def drop_noise_feature(w, dA, f=26, compensate=True, n=40000, seed=0):
    """f=26 is pure white noise (autocorrelation≈0, class separability≈0, kurtosis≈3); the first layer's weights on it only inject noise into z1.
    Set the f=26 rows of W1 to 0 for all t. With compensate=True, also convert "the h1 mean raised by the noise through the convexity of the ReLU"
    into b2, so that the input mean seen by the second layer is the same as in the original network (only unlabelled activation means are used)."""
    w2 = copy(w); rows = np.arange(81) * 28 + f
    w2['W1'][rows] = 0
    if compensate:
        rng = np.random.default_rng(seed); idx = np.sort(rng.choice(len(dA), n, replace=False))
        d = forward(w, dA, idx)['h1'].mean(0) - forward(w2, dA, idx)['h1'].mean(0)
        w2['b2'] = w2['b2'] + d @ w['W2']
    return w2

def prior_correct(w, logits, y):
    """Shift each class's output bias once by log(class frequency / mean predicted probability), correcting the prior mismatch of the output-layer biases;
    [SOS]/[EOS] never occur as labels; their biases are set to -20. Only one step, no iteration."""
    w2 = copy(w)
    pr = np.exp(logits - logits.max(1, keepdims=True)); pr /= pr.sum(1, keepdims=True)
    qbar = pr.mean(0); pi = np.bincount(y, minlength=42) / len(y)
    w2['b3'][:40] += np.log(pi[:40] / qbar[:40]).astype(np.float32)
    w2['b3'][40:] = -20.0
    return w2
```

# Diagnosis 8: far context. (a) data: linear separability eta2 for the centre-frame label of the features lag frames from the centre (the top few dims), decaying with lag;

def eta2_at(f, lag):
    s = dA.utt[max(lag, 0): N + min(lag, 0)] == dA.utt[max(-lag, 0): N - max(lag, 0)]
    yy = y[max(-lag, 0): N - max(lag, 0)][s]; xx = x[max(lag, 0): N + min(lag, 0), f][s]

    zz = z(v); print('%-16s median %.3f  max robust-z %.1f (unit %d)  min robust-z %.1f' % (name, np.median(v), zz.max(), zz.argmax(), zz.min()))
